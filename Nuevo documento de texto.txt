DROP SCHEMA IF EXISTS roguechess CASCADE;
CREATE SCHEMA IF NOT EXISTS roguechess;
SET search_path TO roguechess;

CREATE TABLE IF NOT EXISTS ficha (
    nombre VARCHAR(10)  PRIMARY KEY,
    dmg INTEGER NOT NULL CHECK (dmg >= 0),
    vidamax INTEGER NOT NULL CHECK (vidamax > 0)
);


CREATE TABLE roguechess.puntuacion_equipo (
    nombre_jugador VARCHAR(50) PRIMARY KEY,
    equipo VARCHAR(200) NOT NULL,
    puntos INTEGER NOT NULL CHECK (puntos >= 0),
    fecha TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);
